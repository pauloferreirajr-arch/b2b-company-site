resumo_vendas.py · exemplo
import pandas as pd

def resumo_vendas(arquivo):
    df = pd.read_excel(arquivo)
    return df.groupby("produto")["valor"].sum()

-- e o equivalente em SQL
SELECT produto, SUM(valor) AS total
FROM vendas
GROUP BY produto
ORDER BY total DESC;
